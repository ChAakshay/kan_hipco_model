4. Python Implementation for KAN (pykan)

Use this script to train a Kolmogorov-Arnold Network and extract symbolic rate equations:

import torch
import pandas as pd
from kan import KAN
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler

# 1. Load CSV data (test1 data or online multi-parameter dataset)
df = pd.read_csv("CO2_to_CO_test1_data.csv")

# Input features and target
X = df[["Flow_LPM", "Pressure_Bar", "MW_Power_pct", "Time_min"]].values
y = df[["Conversion_pct"]].values

scaler_X = MinMaxScaler(feature_range=(-1, 1))
scaler_y = MinMaxScaler(feature_range=(-1, 1))
X_scaled = scaler_X.fit_transform(X)
y_scaled = scaler_y.fit_transform(y)

X_train, X_test, y_train, y_test = train_test_split(X_scaled, y_scaled, test_size=0.2, random_state=42)

dataset = {
    'train_input': torch.tensor(X_train, dtype=torch.float32),
    'train_label': torch.tensor(y_train, dtype=torch.float32),
    'test_input': torch.tensor(X_test, dtype=torch.float32),
    'test_label': torch.tensor(y_test, dtype=torch.float32)
}

# 2. Build KAN: 4 inputs -> 3 hidden B-spline nodes -> 1 output
model = KAN(width=[4, 3, 1], grid=5, k=3, seed=42)

# 3. Fit model
model.fit(dataset, opt="Adam", steps=50, lr=0.01)

# 4. Prune and extract symbolic mathematical law
model.prune()
formula = model.symbolic_formula()
print("Discovered Rate Law:")
print(formula[0][0])
